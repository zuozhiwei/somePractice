```java
package edu.campus.treehole;

import java.time.LocalDateTime;
import java.time.ZoneOffset;
import java.util.List;
import java.util.function.Function;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.namedparam.MapSqlParameterSource;
import org.springframework.jdbc.core.namedparam.NamedParameterJdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

@Service
public class PostPages {
    private final NamedParameterJdbcTemplate jdbc;

    public PostPages(NamedParameterJdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public Page<PostController.PostView> list(String category, String q, int limit, String cursor) {
        if (limit < 1 || limit > 50 || q.length() > 100 ||
                !(category.isEmpty() || List.of("DAILY", "HELP", "QUESTION").contains(category))) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "分页或筛选参数无效");
        }
        var params = new MapSqlParameterSource()
                .addValue("category", category).addValue("q", q).addValue("take", limit + 1);
        var sql = new StringBuilder("""
                SELECT id, title, body, category, created_at FROM posts
                WHERE status = 'VISIBLE'
                  AND (:category = '' OR category = :category)
                  AND (:q = '' OR LOCATE(:q, title) > 0 OR LOCATE(:q, body) > 0)
                """);
        if (cursor != null) {
            var after = PageCursor.decode(cursor);
            params.addValue("time", after.createdAt()).addValue("id", after.id());
            sql.append(" AND (created_at < :time OR (created_at = :time AND id < :id))");
        }
        sql.append(" ORDER BY created_at DESC, id DESC LIMIT :take");
        var rows = jdbc.query(sql.toString(), params, (row, index) -> {
            var id = row.getLong("id");
            var time = row.getObject("created_at", LocalDateTime.class);
            return new Row(id, time, new PostController.PostView(
                    Long.toString(id), row.getString("title"), row.getString("body"),
                    row.getString("category"), time.toInstant(ZoneOffset.UTC).toString()));
        });
        return takePage(rows, limit, Row::view, row -> new PageCursor(row.time(), row.id()));
    }

    static <R, T> Page<T> takePage(List<R> rows, int limit, Function<R, T> view,
                                  Function<R, PageCursor> position) {
        var count = Math.min(rows.size(), limit);
        var next = rows.size() > limit ? position.apply(rows.get(count - 1)).encode() : null;
        return new Page<>(rows.subList(0, count).stream().map(view).toList(), next);
    }

    private record Row(long id, LocalDateTime time, PostController.PostView view) {}
    public record Page<T>(List<T> items, String nextCursor) {}
}
```

```js
const listPager = { stamp: 0, items: [], next: null, params: {}, busy: false };
const commentPager = { stamp: 0, items: [], next: null, params: {}, busy: false };

function resetPager(pager, params = {}) {
  Object.assign(pager, { stamp: pager.stamp + 1, items: [], next: null, params, busy: false });
}

async function readPage(pager, path, button, append, render) {
  if (pager.busy || (append && !pager.next)) return;
  const stamp = pager.stamp;
  const params = new URLSearchParams({ limit: '20', ...pager.params });
  if (append) params.set('cursor', pager.next);
  pager.busy = true;
  button.disabled = true;
  try {
    const data = await api(path + '?' + params);
    if (stamp !== pager.stamp) return;
    pager.items = append ? pager.items.concat(data.items) : data.items;
    pager.next = data.nextCursor;
    render(pager.items);
    button.hidden = !pager.next;
  } catch (error) {
    if (stamp === pager.stamp) {
      $("#feedback").textContent = error.message + "，可用加载按钮重试。";
      button.hidden = false;
    }
  } finally {
    if (stamp === pager.stamp) {
      pager.busy = false;
      button.disabled = false;
    }
  }
}
```

### 操作步骤

1. 在 `server/src/main/java/edu/campus/treehole` 新建 `PageCursor.java`。这里的时间类型保持和数据库实体一致，示例使用以 UTC 写入的 `LocalDateTime`。Base64 URL 编码只是让文本容易放进地址；任何传入值都仍要验证。

```java
package edu.campus.treehole;

import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.Base64;
import org.springframework.http.HttpStatus;
import org.springframework.web.server.ResponseStatusException;

record PageCursor(LocalDateTime createdAt, long id) {
    String encode() {
        var value = createdAt + "|" + id;
        return Base64.getUrlEncoder().withoutPadding()
                .encodeToString(value.getBytes(StandardCharsets.UTF_8));
    }

    static PageCursor decode(String value) {
        try {
            if (value.length() > 160) throw new IllegalArgumentException();
            var decoded = new String(Base64.getUrlDecoder().decode(value), StandardCharsets.UTF_8);
            var parts = decoded.split("\\|", -1);
            if (parts.length != 2) throw new IllegalArgumentException();
            var time = LocalDateTime.parse(parts[0]);
            var id = Long.parseLong(parts[1]);
            if (id <= 0 || time.getYear() < 1000 || time.getYear() > 9999) {
                throw new IllegalArgumentException();
            }
            return new PageCursor(time, id);
        } catch (RuntimeException invalid) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "分页位置无效");
        }
    }
}
```
