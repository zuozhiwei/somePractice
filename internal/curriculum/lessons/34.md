### 一份能运行的投递器

把下面代码保存为 `server/src/main/java/edu/campus/treehole/ReplyDispatch.java`。调度器和事务服务分在两个对象上，Spring 才能拦截方法调用、在失败时回滚。贴好后只在已有启动类上加 `@EnableScheduling`，对应 import 是 `org.springframework.scheduling.annotation.EnableScheduling`。如果已有该注解就保留一次。

```java
package edu.campus.treehole;

import java.util.concurrent.atomic.AtomicBoolean;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.dao.DataAccessException;
import org.springframework.dao.DataAccessResourceFailureException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class ReplyDispatch {
    private final JdbcTemplate jdbc;
    private final AtomicBoolean failNext;

    public ReplyDispatch(JdbcTemplate jdbc,
                         @Value("${treehole.replies.drill-fail-first:false}") boolean drill) {
        this.jdbc = jdbc;
        this.failNext = new AtomicBoolean(drill);
    }

    @Transactional
    public boolean deliverOne() {
        var due = jdbc.query("""
            SELECT id, comment_id, recipient_id
            FROM reply_events
            WHERE status='PENDING' AND next_at<=UTC_TIMESTAMP(6)
            ORDER BY next_at,id LIMIT 1 FOR UPDATE
            """, (row, i) -> new Event(row.getLong(1), row.getLong(2), row.getLong(3)));
        if (due.isEmpty()) return false;
        var event = due.get(0);
        try {
            Integer visible = jdbc.queryForObject("""
                SELECT COUNT(*) FROM comments c JOIN posts p ON p.id=c.post_id
                WHERE c.id=? AND p.owner_id=? AND p.status='VISIBLE'
                """, Integer.class, event.commentId(), event.recipientId());
            if (visible == null || visible == 0) {
                jdbc.update("UPDATE reply_events SET status='CANCELLED' WHERE id=?", event.id());
                return true;
            }
            jdbc.update("""
                INSERT INTO reply_notifications (comment_id, recipient_id)
                VALUES (?, ?)
                ON DUPLICATE KEY UPDATE read_at=reply_notifications.read_at
                """, event.commentId(), event.recipientId());
            if (failNext.getAndSet(false)) {
                throw new DataAccessResourceFailureException("drill");
            }
            jdbc.update("UPDATE reply_events SET status='DONE', last_error=NULL WHERE id=?", event.id());
            return true;
        } catch (DataAccessException error) {
            throw new RetryDelivery(event.id(), error);
        }
    }

    @Transactional
    public void postpone(long id) {
        jdbc.update("""
            UPDATE reply_events
            SET status=IF(attempts>=2, 'FAILED', 'PENDING'),
                next_at=TIMESTAMPADD(SECOND, IF(attempts=0, 5, 30), UTC_TIMESTAMP(6)),
                attempts=attempts+1, last_error='TEMPORARY_DATABASE_ERROR'
            WHERE id=? AND status='PENDING'
            """, id);
    }

    private record Event(long id, long commentId, long recipientId) {}

    static final class RetryDelivery extends RuntimeException {
        final long eventId;
        RetryDelivery(long eventId, DataAccessException cause) {
            super(cause);
            this.eventId = eventId;
        }
    }
}

@Component
@ConditionalOnProperty(name="treehole.replies.enabled", havingValue="true", matchIfMissing=true)
class ReplyDispatchWorker {
    private static final org.slf4j.Logger LOG = LoggerFactory.getLogger(ReplyDispatchWorker.class);
    private final ReplyDispatch dispatch;

    ReplyDispatchWorker(ReplyDispatch dispatch) {
        this.dispatch = dispatch;
    }

    @Scheduled(fixedDelayString = "${treehole.replies.delay-ms:3000}")
    public void collect() {
        for (int i = 0; i < 20; i++) {
            try {
                if (!dispatch.deliverOne()) return;
            } catch (ReplyDispatch.RetryDelivery retry) {
                LOG.warn("reply_dispatch_retry event_id={}", retry.eventId);
                try {
                    dispatch.postpone(retry.eventId);
                } catch (DataAccessException unavailable) {
                    LOG.warn("reply_dispatch_reschedule_unavailable");
                    return;
                }
            } catch (DataAccessException unavailable) {
                LOG.warn("reply_dispatch_lookup_unavailable");
                return;
            }
        }
    }
}
```

试验配置只在本机临时加到 Maven 启动参数 `-Dspring-boot.run.arguments="--treehole.replies.enabled=false"`，表示暂停消费者。恢复时去掉参数。`--treehole.replies.drill-fail-first=true` 会在**本次进程的第一条实际投递**写入通知后触发一次异常，以便看到回滚与补跑。验收后撤掉这个参数，发布配置不使用。

```java
boolean markRead(long ownerId, long commentId) {
    jdbc.update("""
        INSERT INTO reply_notifications (comment_id, recipient_id, read_at)
        SELECT c.id, p.owner_id, UTC_TIMESTAMP(6)
        FROM comments c JOIN posts p ON p.id=c.post_id
        WHERE c.id=? AND p.owner_id=? AND p.status='VISIBLE' AND c.owner_id<>p.owner_id
        ON DUPLICATE KEY UPDATE read_at=COALESCE(reply_notifications.read_at, UTC_TIMESTAMP(6))
        """, commentId, ownerId);
    Integer visible = jdbc.queryForObject("""
        SELECT COUNT(*) FROM comments c JOIN posts p ON p.id=c.post_id
        WHERE c.id=? AND p.owner_id=? AND p.status='VISIBLE' AND c.owner_id<>p.owner_id
        """, Integer.class, commentId, ownerId);
    return visible != null && visible > 0;
}
```
